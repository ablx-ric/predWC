\section{Motivación y Datos}

% ══════════════════════════════════════════════════════════════════
\begin{frame}{Motivación y problema}
  \textbf{Problema:} predecir el ganador de cada partido eliminatorio
  del Mundial 2026 (48 selecciones, 3 rondas KO).\\[0.3cm]

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Desafíos:}
      \begin{itemize}\footnotesize
        \item Datos limitados entre selecciones
        \item Desbalance de clases ($\sim$23\% empates)
        \item Sin \emph{leakage} temporal
        \item Scores exactos para visualización
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Solución:}
      \begin{itemize}\footnotesize
        \item Stacking ensemble (RF+XGB+MLP$\to$LR)
        \item Dixon-Coles $\tau$ para scores
        \item Split temporal 80/20
        \item 8\,204 partidos históricos
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.3cm}
  \begin{exampleblock}{Objetivo}
    Accuracy $>$ 60\% en fase eliminatoria, con calibración confiable
    para apoyar análisis táctico.
  \end{exampleblock}
\end{frame}

\begin{frame}{Dixon-Coles: Predicción de scores}

  \textbf{Problema:} Poisson subestima empates 0-0 y 1-1.

  \vspace{0.2cm}
  \textbf{Solución:} corrección $\tau$ (Dixon \& Coles, 1997):

  \[
  P(X{=}x, Y{=}y) \;=\; \tau_{\rho}(x,y)\;
  \frac{e^{-\lambda}\lambda^x}{x!}\;
  \frac{e^{-\mu}\mu^y}{y!}
  \]

  \begin{itemize}
    \item $\rho = -0.13$: corrige correlación en scores bajos
    \item $\tau$ aplica a $(0,0), (1,0), (0,1), (1,1)$
    \item $\lambda/\mu = 10^{\Delta\text{ELO}/400}$ desde ELO
  \end{itemize}

  \vspace{0.2cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Matriz analítica}
      \begin{itemize}\small
        \item PMF Poisson $11\times11$
        \item $\tau$ + renormalización
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Monte Carlo}
      \begin{itemize}\small
        \item 50\,000 muestras
        \item Ranking de scores
        \item Goal ratio desde ELO
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
