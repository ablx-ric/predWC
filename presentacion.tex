\section{Motivación y Datos}

% ══════════════════════════════════════════════════════════════════
\begin{frame}{Motivación y problema}
  \textbf{Problema:} predecir el ganador de cada partido eliminatorio
  del Mundial 2026 (48 selecciones, 3 rondas KO).\\[0.3cm]

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Desafíos:}
      \begin{itemize}\footnotesize
        \item Datos limitados entre selecciones
        \item Desbalance de clases ($\sim$23\% empates)
        \item Sin \emph{leakage} temporal
        \item Scores exactos para visualización
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Solución:}
      \begin{itemize}\footnotesize
        \item Stacking ensemble (RF+XGB+MLP$\to$LR)
        \item Dixon-Coles $\tau$ para scores
        \item Split temporal 80/20
        \item 8\,204 partidos históricos
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.3cm}
  \begin{exampleblock}{Objetivo}
    Accuracy $>$ 60\% en fase eliminatoria, con calibración confiable
    para apoyar análisis táctico.
  \end{exampleblock}
\end{frame}

% ══════════════════════════════════════════════════════════════════
\begin{frame}{Dixon-Coles: Predicción de scores}

  \textbf{Problema:} Poisson subestima empates 0-0 y 1-1.

  \vspace{0.2cm}
  \textbf{Solución:} corrección $\tau$ (Dixon \& Coles, 1997):

  \[
  P(X{=}x, Y{=}y) \;=\; \tau_{\rho}(x,y)\;
  \frac{e^{-\lambda}\lambda^x}{x!}\;
  \frac{e^{-\mu}\mu^y}{y!}
  \]

  \begin{itemize}
    \item $\rho = -0.13$: corrige correlación en scores bajos
    \item $\tau$ aplica a $(0,0), (1,0), (0,1), (1,1)$
    \item $\lambda/\mu = 10^{\Delta\text{ELO}/400}$ desde ELO
  \end{itemize}

  \vspace{0.2cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Matriz analítica}
      \begin{itemize}\small
        \item PMF Poisson $11\times11$
        \item $\tau$ + renormalización
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Monte Carlo}
      \begin{itemize}\small
        \item 50\,000 muestras
        \item Ranking de scores
        \item Goal ratio desde ELO
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{NLP}

% ══════════════════════════════════════════════════════════════════
\begin{frame}{Integración de NLP}

  \begin{exampleblock}{24 features NLP}
    \centering
    {\small
    \begin{tabular}{ll}
      \toprule
      \textbf{Tipo} & \textbf{Features} \\
      \midrule
      News PCA (11 comp. $\times$ 2 equipos) & \code{home\_news\_pc1..11}, \code{away\_news\_pc1..11} \\
      YouTube sentiment ($\times$ 2 equipos)  & \code{home\_yt\_sent}, \code{away\_yt\_sent} \\
      \bottomrule
    \end{tabular}
    }
  \end{exampleblock}

  \vspace{0.2cm}
  \textbf{Problema:} las 24 features existen solo para los 32 equipos
  del Mundial actual, no para los partidos históricos (2018--2024).

  \vspace{0.15cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Estado actual}
      \begin{itemize}\small
        \item Training: NLP = \texttt{0.0} en todos los partidos
        \item Meta-modelo aprende solo de 9 probabilidades base
        \item Inferencia: valores NLP reales, pero el modelo no sabe usarlos
        \item Impacto: $\sim$1pp de ruido (no mejora consistente)
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Arquitectura ideal}
      \begin{itemize}\small
        \item Generar NLP histórico (scrapear noticias de cada fecha)
        \item NLP como entrada directa a \alert{todos} los modelos base
        \item 25 + 24 = 49 features para RF, XGB y MLP
        \item Meta-modelo recibe 9 probs (sin NLP duplicado)
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \centering
  {\small\textbf{Actual:}\; \code{meta\_input = [probas(9) | NLP(24)]}
  \qquad
